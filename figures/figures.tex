\begin{figure*}[t]
  \centering
  \includegraphics[width=\textwidth]{figures/B1_ablation_auc_triptych.pdf}
  \caption{Linear truth decodability requires cross-pair truth
  co-occurrence in the trainable model but not in the toy model.
  Probe AUC at the second-pair position ($x'$, final layer) over
  training, for the normal condition (shared truth bit
  $T \sim \mathrm{Ber}(\rho)$) versus the ablated condition
  (independent $T_1, T_2$; identical marginals, identical mapping
  $g$). \textbf{(a)}~Fully-trained one-layer transformer, $\rho=0.99$
  ($n=5$ seeds); \textbf{(b)}~fully-trained, $\rho=0.8$ ($n=3$);
  \textbf{(c)}~toy model, $\rho=0.8$ ($n=5$). Shading is $\pm 1$ std
  across seeds; dotted line is chance. In the fully-trained model the
  ablated probe never exceeds AUC $0.536$ at any checkpoint of any
  seed, while factual recall stays perfect (mapping accuracy
  $512/512$ in every run); the toy model retains near-ceiling
  decodability ($0.971 \to 0.959$) under the identical intervention.
  Probes: logistic SGD, 5-fold stratified CV, evaluated at
  $p_\mathrm{eval}=0.5$.}
  \label{fig:ablation-auc}
\end{figure*}

\begin{figure*}[t]
  \centering
  \includegraphics[width=0.92\textwidth]{figures/B2_toy_ov_heatmaps_pair.pdf}
  \caption{Toy-model OV circuit under the independent-truth ablation
  ($W = W_o W_v$ at step 2000, seed 0, shared color scale). The
  embedding-space blocks in the $e_x$ rows (negative identity) and the
  reverse-lookup structure collapse in amplitude under ablation, while
  the unembedding-space blocks ($u_y$ rows: memory, self-suppression)
  persist or strengthen --- the probe-decodable residue behind
  Fig.~\ref{fig:ablation-auc}c. See
  Fig.~\ref{fig:ov-amplitude} for the quantitative amplitude analysis.}

\end{figure*}

\begin{figure*}[t]
  \centering
  \includegraphics[width=0.9\textwidth]{figures/B5_ov_amplitude.pdf}
  \caption{The ablation collapses toy OV sub-block amplitudes without
  rotating them. Signed target projection
  $a_B = \langle B, T\rangle / \lVert T \rVert^2$ (left) and Frobenius
  norm (right) of each OV sub-block against its signed-identity
  target at step 2000 (mean $\pm$ SD over 5 seeds per condition).
  Under ablation, negative identity and reverse lookup lose
  $7$--$10\times$ in target projection ($3.59 \to 0.52$ and
  $4.01 \to 0.39$) while memory and self-suppression persist or
  strengthen. Orientation is preserved throughout
  (Fig.~\ref{fig:ov-cosine}), which is why probe AUC alone
  (Fig.~\ref{fig:ablation-auc}c) is an insensitive measure of the
  intervention's mechanistic effect.}
  \label{fig:ov-amplitude}
\end{figure*}

\begin{figure*}[t]
  \centering
  \includegraphics[width=0.85\textwidth]{figures/B3_ov_cosine_4panel.pdf}
  \caption{OV sub-block orientation survives the ablation. Cosine
  similarity between each toy OV sub-block and its signed-identity
  theoretical target over training (mean $\pm$ SD, $n=5$ seeds per
  condition). All four blocks converge to high cosine similarity in
  both conditions; the ablation's effect is almost invisible in this
  metric, in contrast to the $7$--$10\times$ amplitude collapse in
  Fig.~\ref{fig:ov-amplitude}. Cosine measures orientation and is
  invariant to scale, so it alone cannot evaluate an ablation that
  shrinks a circuit without rotating it.}
  \label{fig:ov-cosine}
\end{figure*}

\begin{figure*}[t]
  \centering
  \includegraphics[width=0.9\textwidth]{figures/B4_prob_comparison_rho099.pdf}
  \caption{The ablated fully-trained model converges to the
  Bayes-optimal constant strategy. Probability assigned to the correct
  attribute $g(x')$ on true (left) and false (right) sequences over
  training ($\rho=0.99$, mean $\pm$ SD, $n=5$). The normal model
  learns to modulate its confidence by the truth of the first pair;
  the ablated model always predicts $g(x')$ with confidence
  $\approx \rho$ on both sequence types, exactly as information theory
  predicts when $I(T_1; T_2)=0$.}
  \label{fig:prob-comparison}
\end{figure*}

  \caption{Faithful reproduction of the original paper's Figure~6a at
  two scales. NLL of the correct attribute after $k$ false (F$^k$) or
  true (T$^k$) context sentences (CounterFact P103, native language;
  896 prompts per model; dashed line = no-context median; diamonds =
  medians). The reproduction deliberately retains the original
  pipeline's flaws (attribute priming, answer leakage, unpaired

  $1.52$ on Llama-3-8B); 70B $1.83$, growing with context length

\begin{figure*}[t]
  \centering
  \includegraphics[width=\textwidth]{figures/C2_behavioral_grid.pdf}
  \caption{The behavioral context-truthfulness effect transfers across
  relations. Median paired $\Delta$NLL (false $-$ true context) over a
  $6\times 6$ context-relation $\times$ target-relation grid, under the
  controlled design (paired true/false versions of the same
  context facts, subject exclusion, anti-priming exclusion; 64 pairs
  per cell). Positive values mean false context hurts recall of the
  target fact. The within-relation (diagonal) effect is significant in
  all six relations for both models (Wilcoxon $p<0.05$); off-diagonal
  transfer is significant in 29/30 cells for 70B versus 27/30 for 8B,
  with mean off-diagonal $\Delta$NLL $0.51$ versus $0.32$ nats --- the
  effect is more relation-general at 70B.}
  \label{fig:behavioral-grid}
\end{figure*}
